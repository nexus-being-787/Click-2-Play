Dust & Footstep Debris Particle Emitter class DebrisSystem { constructor(scene) { this.scene = scene; this.maxParticles = 60; this.particles = []; const geo = new THREE.BufferGeometry(); this.positions = new Float32Array(this.maxParticles * 3); this.opacities = new Float32Array(this.maxParticles); geo.setAttribute("position", new THREE.BufferAttribute(this.positions, 3)); this.mat = new THREE.PointsMaterial({ color: 0x9e8a78, size: 0.18, transparent: true, opacity: 0.6, blending: THREE.NormalBlending }); this.mesh = new THREE.Points(geo, this.mat); this.scene.add(this.mesh); for (let i = 0; i < this.maxParticles; i++) { this.particles.push({ x: 0, y: -10, z: 0, vx: 0, vy: 0, vz: 0, life: 0, maxLife: 0.4 }); } } emit(x, y, z) { for (let i = 0; i < 3; i++) { const p = this.particles.find(pt => pt.life <= 0); if (!p) break; p.x = x + (Math.random() - 0.5) * 0.3; p.y = y + 0.05; p.z = z + (Math.random() - 0.5) * 0.2; p.vx = (Math.random() - 0.5) * 1.5; p.vy = Math.random() * 1.2 + 0.5; p.vz = Math.random() * 2.0 + 1.0; p.life = p.maxLife; } } update(delta) { const pos = this.positions; for (let i = 0; i < this.maxParticles; i++) { const p = this.particles[i]; if (p.life > 0) { p.life -= delta; p.x += p.vx * delta; p.y += p.vy * delta; p.z += p.vz * delta; p.vy -= 4.0 * delta; // Gravity on dirt flecks pos[i * 3 + 0] = p.x; pos[i * 3 + 1] = p.y; pos[i * 3 + 2] = p.z; } else { pos[i * 3 + 1] = -100; } } this.mesh.geometry.attributes.position.needsUpdate = true; } } // 2. High-Speed Wind Tunnel Lines (Speed Sensation) class SpeedStreakSystem { constructor(scene) { this.scene = scene; this.count = 45; const geo = new THREE.BufferGeometry(); const pos = new Float32Array(this.count * 6); // 2 vertices per line segment for (let i = 0; i < this.count; i++) { const x = (Math.random() - 0.5) * 12; const y = Math.random() * 5 + 0.5; const z = -Math.random() * 30; const len = Math.random() * 2.5 + 1.5; pos[i * 6 + 0] = x; pos[i * 6 + 1] = y; pos[i * 6 + 2] = z; pos[i * 6 + 3] = x; pos[i * 6 + 4] = y; pos[i * 6 + 5] = z - len; } geo.setAttribute("position", new THREE.BufferAttribute(pos, 3)); this.mat = new THREE.LineBasicMaterial({ color: 0xcccccc, transparent: true, opacity: 0.25 }); this.lines = new THREE.LineSegments(geo, this.mat); this.scene.add(this.lines); } update(playerZ, speed) { // Scale streak visibility with current player velocity this.mat.opacity = Math.min(0.55, (speed - 28) / 45); const pos = this.lines.geometry.attributes.position.array; for (let i = 0; i < this.count; i++) { if (pos[i * 6 + 2] > playerZ + 5) { const x = (Math.random() - 0.5) * 10; const y = Math.random() * 5 + 0.5; const z = playerZ - 25 - Math.random() * 25; const len = 2.0 + (speed / 15); pos[i * 6 + 0] = x; pos[i * 6 + 1] = y; pos[i * 6 + 2] = z; pos[i * 6 + 3] = x; pos[i * 6 + 4] = y; pos[i * 6 + 5] = z - len; } } this.lines.geometry.attributes.position.needsUpdate = true; } } // 3. Touch / Swipe Controller for Mobile Devices function setupTouchControls() { let touchStartX = 0; let touchStartY = 0; const minSwipeDist = 35; window.addEventListener("touchstart", (e) => { touchStartX = e.changedTouches[0].screenX; touchStartY = e.changedTouches[0].screenY; }, { passive: true }); window.addEventListener("touchend", (e) => { if (Engine.state !== GAME_STATE.RUNNING) return; const diffX = e.changedTouches[0].screenX - touchStartX; const diffY = e.changedTouches[0].screenY - touchStartY; if (Math.abs(diffX) > Math.abs(diffY)) { // Horizontal Swipes if (diffX > minSwipeDist && Engine.currentLane < 2) { Engine.currentLane++; Engine.targetX = (Engine.currentLane - 1) * Engine.laneWidth; } else if (diffX < -minSwipeDist && Engine.currentLane > 0) { Engine.currentLane--; Engine.targetX = (Engine.currentLane - 1) * Engine.laneWidth; } } else { // Vertical Swipes if (diffY < -minSwipeDist && !Engine.isJumping && !Engine.isSliding) { Engine.isJumping = true; Engine.jumpVelocity = 17.5; Engine.sound.playJump(); } else if (diffY > minSwipeDist && !Engine.isSliding) { Engine.isSliding = true; Engine.slideTimeRemaining = 0.65; Engine.sound.playSlide(); if (Engine.isJumping) Engine.jumpVelocity = -22; } } }, { passive: true }); } // 4. Hook Systems into World Lifecycle const originalWorldInit = World.init; World.init = function() { originalWorldInit.call(this); this.debris = new DebrisSystem(this.scene); this.speedStreaks = new SpeedStreakSystem(this.scene); setupTouchControls(); // Hook debris spawner onto running footstep timing const originalAnimate = this.animate; this.animate = (now) => { if (Engine.state === GAME_STATE.RUNNING) { this.debris.update(Math.min((now - this.lastTime) / 1000, 0.1)); this.speedStreaks.update(this.character.root.position.z, Engine.speed); if (!Engine.isJumping && !Engine.isSliding) { const feetY = this.character.root.position.y; this.debris.emit( this.character.root.position.x, feetY, this.character.root.position.z ); } } originalAnimate.call(this, now); }; };
